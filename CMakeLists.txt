cmake_minimum_required(VERSION 3.20)
project(snewsware LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(snewsware WIN32 src/main.cpp src/snewsware.rc)
target_compile_definitions(snewsware PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(snewsware PRIVATE user32 gdi32 dwmapi shell32)
if(MSVC)
  target_compile_options(snewsware PRIVATE /W4 /permissive- /EHsc)
  target_link_options(snewsware PRIVATE /SUBSYSTEM:WINDOWS)
endif()
install(TARGETS snewsware RUNTIME DESTINATION .)
